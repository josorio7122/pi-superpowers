```ts
import { Value } from "@sinclair/typebox/value";
import { TodoDetailsSchema, type TodoItem } from "./schema.js";

/**
 * Accepts both pi's real SessionMessageEntry shape and a flat legacy shape:
 *
 * Real pi shape (verified from SessionMessageEntry + ToolResultMessage):
 *   { type: "message", message: { role: "toolResult", toolName, details, ... } }
 *
 * Legacy test-fixture shape (kept for backward compat):
 *   { tool, details }
 */
export type SessionEntryLike = {
	type?: string;
	message?: {
		role?: string;
		toolName?: string;
		details?: unknown;
	};
	tool?: string;
	details?: unknown;
};

function entryToolName(entry: SessionEntryLike): string | undefined {
	if (entry.message?.role === "toolResult") return entry.message.toolName;
	if (entry.tool) return entry.tool;
	return undefined;
}

function entryDetails(entry: SessionEntryLike): unknown {
	if (entry.message?.role === "toolResult") return entry.message.details;
	return entry.details;
}

export function reconstructTodos(entries: SessionEntryLike[]): TodoItem[] {
	for (let i = entries.length - 1; i >= 0; i--) {
		const entry = entries[i];
		if (!entry) continue;
		if (entryToolName(entry) !== "superpowers_todo") continue;
		const details = entryDetails(entry);
		if (!Value.Check(TodoDetailsSchema, details)) return [];
		return details.todos;
	}
	return [];
}
```

```ts
import { readdir, readFile, stat } from "node:fs/promises";
import { basename, join } from "node:path";
import { vendorAgentsDir, vendorSkillsDir } from "../common/paths.js";
import { type AgentFrontmatterLike, parseAgentMarkdown } from "./frontmatter.js";

export type LoadAgentsOptions = {
	agentsDir?: string;
	skillsDir?: string;
};

export type LoadedAgents = {
	agents: AgentFrontmatterLike[];
	diagnostics: Array<{ file: string; reason: string }>;
};

async function loadAgentsFromAgentsDir(dir: string): Promise<LoadedAgents> {
	const agents: AgentFrontmatterLike[] = [];
	const diagnostics: Array<{ file: string; reason: string }> = [];
	let entries: string[];
	try {
		entries = await readdir(dir);
	} catch {
		return { agents, diagnostics: [{ file: dir, reason: "agents dir missing" }] };
	}
	for (const name of entries) {
		if (!name.endsWith(".md")) continue;
		const path = join(dir, name);
		try {
			const src = await readFile(path, "utf8");
			const parsed = parseAgentMarkdown(src);
			if (!parsed) {
				diagnostics.push({ file: name, reason: "missing or invalid frontmatter" });
				continue;
			}
			agents.push(parsed);
		} catch (err) {
			diagnostics.push({ file: name, reason: `read failed: ${(err as Error).message}` });
		}
	}
	return { agents, diagnostics };
}

function firstNonEmptyLine(body: string): string {
	for (const raw of body.split("\n")) {
		const trimmed = raw.replace(/^#+\s*/, "").trim();
		if (trimmed.length > 0) return trimmed;
	}
	return "";
}

async function synthesizeAgentFromPromptFile(path: string): Promise<AgentFrontmatterLike | null> {
	try {
		const body = (await readFile(path, "utf8")).trim();
		if (!body) return null;
		const name = basename(path).replace(/-prompt\.md$/, "").replace(/\.md$/, "");
		if (!name) return null;
		const description = firstNonEmptyLine(body) || `Synthesized from ${basename(path)}`;
		return { name, body, description };
	} catch {
		return null;
	}
}

async function walkSkillPromptFiles(dir: string): Promise<string[]> {
	const out: string[] = [];
	let entries: string[];
	try {
		entries = await readdir(dir);
	} catch {
		return out;
	}
	for (const entry of entries) {
		const p = join(dir, entry);
		try {
			const s = await stat(p);
			if (s.isDirectory()) {
				out.push(...(await walkSkillPromptFiles(p)));
			} else if (s.isFile() && entry.endsWith("-prompt.md")) {
				out.push(p);
			}
		} catch {
			// ignore unreadable entries
		}
	}
	return out;
}

export async function loadAgents(opts: LoadAgentsOptions = {}): Promise<LoadedAgents> {
	const agentsDir = opts.agentsDir ?? vendorAgentsDir();
	const skillsDir = opts.skillsDir ?? vendorSkillsDir();

	const primary = await loadAgentsFromAgentsDir(agentsDir);
	const seenNames = new Set(primary.agents.map((a) => a.name));

	const promptFiles = await walkSkillPromptFiles(skillsDir);
	for (const path of promptFiles) {
		const synth = await synthesizeAgentFromPromptFile(path);
		if (!synth) continue;
		if (seenNames.has(synth.name)) continue;
		seenNames.add(synth.name);
		primary.agents.push(synth);
	}

	return primary;
}

export function findAgent(agents: AgentFrontmatterLike[], name: string): AgentFrontmatterLike | undefined {
	return agents.find((a) => a.name === name);
}
```

```ts
describeIfPi("superpowers_todo add → complete → list round-trip e2e", () => {

		"completing a previously-added item preserves the item in list (no state loss)",
		async () => {
			const result = await runPiE2EWithRetry(
				{
					prompt: [
						'Do this in three steps:',
						'1. Call superpowers_todo with {"action":"add","content":"alpha"}.',
						'2. After seeing the response, call superpowers_todo with {"action":"add","content":"beta"}.',
						'3. After that, call superpowers_todo with {"action":"complete","id":"1"}.',
						'Then print the final response panel exactly.',
					].join("\n"),
					timeoutMs: 180_000,
				},
				{
					attempts: 3,
					// The bug would make step 3 return empty details.todos or an error.
					// After the fix, the completed item must appear AND beta must still be present.
					check: (r) => {
						const todoMarkers = r.markers.filter((m) => m.name === "widget-set");
						// We expect widget-set with lineCount>0 AFTER the complete action —
						// widgets only stay populated if prior state reconstructed.
						return todoMarkers.some((m) => (m.payload as { lineCount: number }).lineCount > 0);
					},
				},
			);
			try {
				const widgets = result.markers.filter((m) => m.name === "widget-set");
				// At least one widget-set after the complete should have > 0 lines.
				expect(widgets.some((w) => (w.payload as { lineCount: number }).lineCount > 0)).toBe(true);
			} finally {
				await result.cleanup();
			}
		},
		600_000,
	);
```
